import cadquery as cq
# Kabinet: ön yüz Y=0 (ayaklar), arka Y=40. Yan profil (y,z): ön yükseklik 10, eğik yüz y=0..30 (z=10->40), üst düz şerit y=30..40
GEN=60.0   # X genişlik
DER=40.0   # Y derinlik
AYAK=15.0  # ayak genişliği (iki yanda)
OLUK_DER=20.0  # TAHMİN: oluk derinliği (ön yüzden), arka duvar z=30'a kadar (çizimden)
prof=[(0,0),(DER,0),(DER,40),(30,40),(0,10)]
govde=cq.Workplane("YZ").polyline(prof).close().extrude(GEN)
# ortadaki oluk: x 15..45, y -1..20, eğik yüzü delerek üstten açık
oluk=cq.Workplane("XY").box(GEN-2*AYAK,OLUK_DER+1,60,centered=False).translate((AYAK,-1,-1))
result=govde.cut(oluk)
# BEKLENEN SINIR KUTUSU: 60 x 40 x 40 mm
