import cadquery as cq
# TAHMIN olcular (mm), olcusuz cizim
L=80.0; B=50.0; H=40.0   # X, Y, Z TAHMIN
R=8.0; T=5.0             # kose yaricapi, cidar TAHMIN
SLOT_W=26.0; SLOT_X=24.0 # on cidardaki U kanal, X merkezi TAHMIN
BOSS_OD=22.0; BOSS_ID=12.0; BOSS_H=22.0 # TAHMIN
HOLE_D=8.0; HOLE_Z=26.0  # TAHMIN
body=cq.Workplane("XY").box(L,B,H,centered=False).edges("|Z").fillet(R)
cav=cq.Workplane("XY").box(L-2*T,B-2*T,H,centered=False).translate((T,T,T)).edges("|Z").fillet(R-T)
body=body.cut(cav)
# on cidarda ustten acik kanal
slot=cq.Workplane("XY").box(SLOT_W,2*T+2,H-T+1,centered=False).translate((SLOT_X-SLOT_W/2,-1,T))
body=body.cut(slot)
# ic silindirik bos (tabandan yukselen halka)
boss=cq.Workplane("XY").center(SLOT_X,T+8).circle(BOSS_OD/2).circle(BOSS_ID/2).extrude(BOSS_H).translate((0,0,T-0.5))
body=body.union(boss)
# X yonunde delikler (iki uc cidar)
h=cq.Workplane("YZ").center(B/2,HOLE_Z).circle(HOLE_D/2).extrude(L)
result=body.cut(h)
# BEKLENEN SINIR KUTUSU: 80 x 50 x 40 mm
