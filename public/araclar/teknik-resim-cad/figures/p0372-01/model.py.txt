import cadquery as cq
# Olcusuz cizim: hepsi TAHMIN (mm). Eksenler: X = göbek ekseni (asagi-saga), Y = yukari-saga, Z yukari. Bakis 1,-1,1
HUB_D=40.0      # TAHMIN gobek dis capi
HUB_L=30.0      # TAHMIN gobek uzunlugu (X)
BORE_D=20.0     # TAHMIN delik capi
KEY_W=6.0; KEY_H=3.0    # TAHMIN kama yuvasi (genislik, ek derinlik)
POST_X=16.0     # TAHMIN direk X derinligi
POST_Y=20.0     # TAHMIN direk Y genisligi
POST_TOP=80.0   # TAHMIN direk tepe Z
HUB_Z=HUB_D/2   # gobek ekseni Z=20, Y=0
WIN_Y=12.0; WIN_Z0=36.0; WIN_Z1=72.0   # TAHMIN pencere
TOPHOLE_D=6.0; TOPHOLE_DEPTH=10.0     # TAHMIN ust delik
hub=cq.Workplane("YZ").center(0,HUB_Z).circle(HUB_D/2).extrude(HUB_L)
post=cq.Workplane("XY").box(POST_X,POST_Y,POST_TOP-HUB_Z,centered=False).translate((0,-POST_Y/2,HUB_Z))
body=hub.union(post)
# delik + kama yuvasi (kama ustte)
bore=cq.Workplane("YZ").workplane(offset=-1).center(0,HUB_Z).circle(BORE_D/2).extrude(HUB_L+2)
key=cq.Workplane("XY").box(HUB_L+2,KEY_W,KEY_H+1,centered=False).translate((-1,-KEY_W/2,HUB_Z+BORE_D/2-1))
# pencere: direkten X boyunca gecen dikdortgen
win=cq.Workplane("XY").box(POST_X+2,WIN_Y,WIN_Z1-WIN_Z0,centered=False).translate((-1,-WIN_Y/2,WIN_Z0))
# ust delik
top=cq.Workplane("XY").workplane(offset=POST_TOP-TOPHOLE_DEPTH).center(POST_X/2,0).circle(TOPHOLE_D/2).extrude(TOPHOLE_DEPTH+1)
result=body.cut(bore).cut(key).cut(win).cut(top)
# BEKLENEN SINIR KUTUSU: 30 x 40 x 80 mm
