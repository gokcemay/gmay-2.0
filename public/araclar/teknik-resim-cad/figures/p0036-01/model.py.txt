import cadquery as cq
# Olculer mm (cizimde var)
X_TOPLAM=60; DERINLIK=40; YUKSEKLIK=40
ALCAK_H=20      # alcak kisim yuksekligi
ALCAK_X=40      # alcak kisim X uzunlugu (60-20)
TEPE_GEN=20     # yuksek kolon genisligi
OYUK_X=20; OYUK_Y=20   # sol-on kose bosluk (alcak kisim derinligi 20)
def box(x0,x1,y0,y1,z0,z1):
    return cq.Workplane("XY").box(x1-x0,y1-y0,z1-z0,centered=False).translate((x0,y0,z0))
alcak=box(0,ALCAK_X,0,DERINLIK,0,ALCAK_H).cut(box(0,OYUK_X,0,OYUK_Y,0,ALCAK_H))  # sol-on oyuk
kolon=box(ALCAK_X,X_TOPLAM,0,DERINLIK,0,YUKSEKLIK)
result=alcak.union(kolon)
