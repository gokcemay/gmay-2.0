import cadquery as cq
# Kavaliyer/kabinet: ön yüz XZ gerçek şekil, derinlik +Y. 
W=80.0; D=40.0
STEP_D=8.0     # alt basamağın derinliği (8+24+8=40 yan ölçülerden; soldaki '10' ile çelişir) TAHMİN
H_LOW=12.0; H_UP=20.0
HOLE_D=10.0; HOLE_X=(20.0,60.0)   # 40 aralık, sağ uçtan 20
NOTCH_X=4.0; NOTCH_Y0=32.0        # sağ arka üst kertik: 4 x 8, TAHMİN (yüksekliği üst basamak)
# 1) Basamaklı profil (Y,Z) -> X boyunca
pts=[(0,0),(D,0),(D,H_UP),(STEP_D,H_UP),(STEP_D,H_LOW),(0,H_LOW)]
body=cq.Workplane("YZ").polyline(pts).close().extrude(W)
# 2) İki Ø10 delik üst basamakta (Y ortası)
yc=(STEP_D+D)/2
for x in HOLE_X:
    body=body.cut(cq.Workplane("XY",origin=(x,yc,H_LOW-1)).circle(HOLE_D/2).extrude(H_UP-H_LOW+2))
# 3) Sağ arka kertik
notch=cq.Workplane("XY").box(NOTCH_X+1,D-NOTCH_Y0+1,H_UP-H_LOW+1,centered=False).translate((W-NOTCH_X,NOTCH_Y0,H_LOW))
result=body.cut(notch)
