import cadquery as cq
DER=40
# L profil (x,z): taban 30 x 20, kolon 10 genis 40 yuksek
L=[(0,0),(30,0),(30,20),(10,20),(10,40),(0,40)]
govde=cq.Workplane("XZ").polyline(L).close().extrude(-DER)
# Sol kol z=0..10, x=-30..0: y=20..40 + x=-30..-20, y=10..20
k1=cq.Workplane("XY").box(30,20,10,centered=False).translate((-30,20,0))
k2=cq.Workplane("XY").box(10,10,10,centered=False).translate((-30,10,0))
result=govde.union(k1).union(k2)
