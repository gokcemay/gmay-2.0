import cadquery as cq
# Olcusuz: TAHMIN (mm)
X=70; Y=40; Z=45          # TAHMIN govde
govde=cq.Workplane("XY").box(X,Y,Z,centered=False)
# Derin kisim (sol-on), tabana yakin biter; TAHMIN
derin=cq.Workplane("XY").workplane(offset=5).rect(18,15,centered=False).extrude(Z).translate((10,10,0))
# Sig kisim (sag-arka), orta derinlikte biter; TAHMIN
sig=cq.Workplane("XY").workplane(offset=Z-30).rect(22,19,centered=False).extrude(31).translate((28,10,0))
result=govde.cut(derin).cut(sig)
